# Cases for the bubblewrap mount planner. See tests/mount_plan_cases.rs for the
# format.

case: a policy with no entries leaves every base mount read-only
bind: /workspace /h/ws rw
bind: /usr /h/usr ro
deny_read: /nowhere
expect: bind ro /usr | bind ro /workspace
expect-tmp: readonly

case: a granted workspace stays writable and a denied file inside it becomes a stub
bind: /workspace /h/ws rw
file: /h/ws/secret
allow_write: /workspace
deny_read: /workspace/secret
expect: bind rw /workspace | stub /workspace/secret

case: a denied directory becomes an empty tmpfs that is remounted read-only last
bind: /workspace /h/ws rw
dir: /h/ws/vault
file: /h/ws/vault/key
allow_write: /workspace
deny_read: /workspace/vault
expect: bind rw /workspace | tmpfs /workspace/vault | remount-ro /workspace/vault

case: ancestors of a protected path are pinned so a rename cannot walk around it
bind: /workspace /h/ws rw
file: /h/ws/a/b/denied
allow_write: /workspace
deny_read: /workspace/a/b/denied
expect: bind rw /workspace | bind rw /workspace/a | bind rw /workspace/a/b | stub /workspace/a/b/denied

case: nothing is pinned when the region is read-only
bind: /workspace /h/ws ro
file: /h/ws/a/b/denied
deny_read: /workspace/a/b/denied
expect: bind ro /workspace | stub /workspace/a/b/denied

case: a file on the way to a denied path is held read-only so it cannot become a directory
bind: /workspace /h/ws rw
file: /h/ws/blocker
allow_write: /workspace
deny_read: /workspace/blocker/inner
expect: bind rw /workspace | bind ro /workspace/blocker

case: one mount per destination whatever lists name it
bind: /workspace /h/ws rw
file: /h/ws/x.env
allow_write: /workspace
deny_read: /workspace/x.env | /workspace/x.env | glob:/workspace/*.env
deny_write: /workspace/x.env
expect: bind rw /workspace | stub /workspace/x.env

case: the plan does not depend on the order of the entries
bind: /workspace /h/ws rw
file: /h/ws/a
file: /h/ws/b
allow_write: /workspace
deny_read: /workspace/b | /workspace/a
expect: bind rw /workspace | stub /workspace/a | stub /workspace/b

case: the plan does not depend on the order of the entries (reversed)
bind: /workspace /h/ws rw
file: /h/ws/a
file: /h/ws/b
allow_write: /workspace
deny_read: /workspace/a | /workspace/b
expect: bind rw /workspace | stub /workspace/a | stub /workspace/b

case: a denied path that does not exist yet gets a guard at its first missing component
bind: /workspace /h/ws rw
allow_write: /workspace
deny_read: /workspace/later/.env
expect: bind rw /workspace | stub /workspace/later
expect-placeholders: /h/ws/later

case: a guard is not needed where the sandbox cannot write
bind: /workspace /h/ws ro
deny_read: /workspace/later/.env
expect: bind ro /workspace
expect-placeholders:

case: a deny_write guard that the worker cannot create is refused
bind: /workspace /h/ws rw
nocreate: /h/ws
allow_write: /workspace
deny_write: /workspace/.git/hooks/pre-commit
expect-error: unguardable-deny-write

case: a deny_read guard that the worker cannot create is reported instead of dropped
bind: /workspace /h/ws rw
nocreate: /h/ws
file: /h/ws/.gitconfig
file: /h/ws/.gitmodules
file: /h/ws/.bashrc
file: /h/ws/.bash_profile
file: /h/ws/.zshrc
file: /h/ws/.zprofile
file: /h/ws/.profile
file: /h/ws/.ripgreprc
file: /h/ws/.mcp.json
dir: /h/ws/.vscode
dir: /h/ws/.idea
dir: /h/ws/.agent/commands
dir: /h/ws/.agent/agents
allow_write: /workspace
deny_read: /workspace/later
expect: bind rw /workspace | bind rw /workspace/.agent
expect-unguarded: /workspace/later

case: a denied symlink protects its target
bind: /workspace /h/ws rw
file: /h/ws/real
link: /h/ws/alias -> real
allow_write: /workspace
deny_read: /workspace/alias
expect: bind rw /workspace | stub /workspace/real

case: an absolute link target is read in the guest, not on the host
bind: /workspace /h/ws rw
file: /h/ws/secret
link: /h/ws/abs -> /workspace/secret
allow_write: /workspace
deny_read: /workspace/abs
expect: bind rw /workspace | stub /workspace/secret

case: a link in the middle of a path is followed
bind: /workspace /h/ws rw
dir: /h/ws/real
file: /h/ws/real/key
link: /h/ws/dir -> real
allow_write: /workspace
deny_read: /workspace/dir/key
expect: bind rw /workspace | bind rw /workspace/real | stub /workspace/real/key

case: a write grant reached through a link is not honoured
bind: /workspace /h/ws rw
dir: /h/ws/real
link: /h/ws/alias -> real
allow_write: /workspace/alias
expect: bind ro /workspace
expect-suppressed: allow_write /workspace/real via-symlink

case: a write grant through an absolute link cannot escape to another mount
bind: /workspace /h/ws rw
bind: /usr /h/usr ro
dir: /h/usr/lib
link: /h/ws/out -> /usr/lib
allow_write: /workspace/out
expect: bind ro /usr | bind ro /workspace
expect-suppressed: allow_write /usr/lib via-symlink

case: a link loop is refused
bind: /workspace /h/ws rw
link: /h/ws/a -> b
link: /h/ws/b -> a
deny_read: /workspace/a
expect-error: symlink-loop

case: a deny inside a private region is refused
bind: /workspace /h/ws rw
deny_read: /proc/self/environ
expect-error: private-region

case: a write allow inside /tmp is refused
bind: /workspace /h/ws rw
allow_write: /tmp/cache
expect-error: private-region

case: allow_write of exactly /tmp opens the tmpfs
bind: /workspace /h/ws rw
allow_write: /tmp
expect: bind ro /workspace
expect-tmp: writable

case: deny_write of /tmp closes the tmpfs again
bind: /workspace /h/ws rw
allow_write: /tmp
deny_write: /tmp
expect: bind ro /workspace
expect-tmp: readonly

case: an allow inside a deny is bound back over the stub
bind: /workspace /h/ws rw
dir: /h/ws/vault/open
file: /h/ws/vault/open/ok
file: /h/ws/vault/inner
deny_read: /workspace/vault
allow_read: /workspace/vault/open
expect: bind ro /workspace | tmpfs /workspace/vault | bind ro /workspace/vault/open | remount-ro /workspace/vault

case: a write grant strictly beneath a hidden directory is dropped, not bound back writable
bind: /workspace /h/ws rw
dir: /h/ws/vault/out
allow_write: /workspace/vault/out
deny_read: /workspace/vault
expect: bind ro /workspace | tmpfs /workspace/vault | remount-ro /workspace/vault
expect-suppressed: allow_write /workspace/vault/out hidden

case: a write grant inside a hidden directory is dropped
bind: /workspace /h/ws rw
dir: /h/ws/vault
allow_write: /workspace/vault
deny_read: /workspace/vault
expect: bind ro /workspace | tmpfs /workspace/vault | remount-ro /workspace/vault
expect-suppressed: allow_write /workspace/vault hidden

case: a write grant on a read-only mount is reported
bind: /workspace /h/ws ro
dir: /h/ws/out
allow_write: /workspace/out
expect: bind ro /workspace
expect-suppressed: allow_write /workspace/out read-only-mount

case: a write deny inside a write grant becomes a read-only bind
bind: /workspace /h/ws rw
dir: /h/ws/out/locked
allow_write: /workspace/out
deny_write: /workspace/out/locked
expect: bind ro /workspace | bind rw /workspace/out | bind ro /workspace/out/locked

case: a write deny on a path that is already read-only adds nothing
bind: /workspace /h/ws rw
dir: /h/ws/frozen
deny_write: /workspace/frozen
expect: bind ro /workspace

case: a write deny pattern that overlaps a grant is refused
bind: /workspace /h/ws rw
allow_write: /workspace
deny_write: glob:/workspace/*.lock
expect-error: unenforceable-write-glob

case: a write deny pattern away from every grant is a no-op
bind: /workspace /h/ws rw
dir: /h/ws/out
allow_write: /workspace/out
deny_write: glob:/workspace/frozen/*.lock
expect: bind ro /workspace | bind rw /workspace/out

case: a read deny pattern is expanded against the tree
bind: /workspace /h/ws rw
file: /h/ws/a.env
file: /h/ws/b.env
file: /h/ws/c.txt
dir: /h/ws/sub
file: /h/ws/sub/d.env
allow_write: /workspace
deny_read: glob:/workspace/*.env
expect: bind rw /workspace | stub /workspace/a.env | stub /workspace/b.env

case: a recursive read deny pattern reaches nested files
bind: /workspace /h/ws rw
file: /h/ws/a.env
dir: /h/ws/sub
file: /h/ws/sub/d.env
allow_write: /workspace
deny_read: glob:/workspace/**/*.env
expect: bind rw /workspace | stub /workspace/a.env | bind rw /workspace/sub | stub /workspace/sub/d.env

case: a write allow pattern grants only what exists now
bind: /workspace /h/ws rw
dir: /h/ws/out1
dir: /h/ws/out2
dir: /h/ws/other
allow_write: glob:/workspace/out*
expect: bind ro /workspace | bind rw /workspace/out1 | bind rw /workspace/out2

case: a pattern whose literal base is a symlink protects the target
bind: /workspace /h/ws rw
dir: /h/ws/real
file: /h/ws/real/k.env
link: /h/ws/alias -> real
allow_write: /workspace
deny_read: glob:/workspace/alias/*.env
expect: bind rw /workspace | bind rw /workspace/real | stub /workspace/real/k.env

case: a pattern follows a symlink inside the walk and denies what it reaches where it lives
bind: /workspace /h/ws rw
bind: /usr /h/usr ro
dir: /h/usr/s
file: /h/usr/s/k.env
link: /h/ws/alias -> /usr/s
allow_write: /workspace
deny_read: glob:/workspace/**/*.env
expect: bind ro /usr | bind rw /workspace | stub /usr/s/k.env

case: denying a base mount hides it
bind: /workspace /h/ws rw
bind: /usr /h/usr ro
deny_read: /workspace
expect: bind ro /usr | tmpfs /workspace | remount-ro /workspace

case: a path that is not in the guest composes nothing
bind: /workspace /h/ws rw
deny_read: /home/someone/.ssh
expect: bind ro /workspace

case: system symlinks are recreated before the binds
bind: /usr /h/usr ro
bind: /workspace /h/ws rw
symlink: /bin usr/bin
deny_read: /nowhere
expect: symlink usr/bin /bin | bind ro /usr | bind ro /workspace
