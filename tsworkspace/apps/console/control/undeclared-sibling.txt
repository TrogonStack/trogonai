import { existsSync } from 'node:fs';
import { join } from 'node:path';

import { expect, it } from 'vitest';

it('is an undeclared sibling that must never be collected', () => {
  const declared = existsSync(
    join(process.env.TEST_SRCDIR ?? '', '_main/tsworkspace/apps/console/src/screens/declared1.test.tsx'),
  );
  console.log(`[control] undeclared sibling declared1.test.tsx collected, present in runfiles: ${declared}`);
  expect.fail('Control: the undeclared sibling declared1.test.tsx was collected.');
});
